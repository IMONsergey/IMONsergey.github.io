    return {
      total: questions.length,
      answered,
      complete: required.length > 0 && required.every(hasAnswer)
    };
  }

  function ensureCurrentQuestion() {
    const questions = visibleQuestions();
    if (!questions.length) return null;
    const current = questionById(state.currentQuestionId);
    if (current) return current;
    const firstUnanswered = questions.find((question) => !hasAnswer(question));
    state.currentQuestionId = (firstUnanswered || questions[0]).id;
    saveState();
    return questionById(state.currentQuestionId);
  }

  function currentIndex() {
    const questions = visibleQuestions();
    const index = questions.findIndex((question) => question.id === state.currentQuestionId);
    return index >= 0 ? index : 0;
  }

  function currentSection() {
    const question = ensureCurrentQuestion();
    return data.sections.find((section) => section.id === question?.section) || data.sections[0];
  }

    clearTimeout(transitionTimer);
    document.onkeydown = null;
    resetButton.hidden = state.screen === "intro" && Object.keys(state.answers).length === 0;

    if (state.screen === "intro") renderIntro();
    else if (state.screen === "test") renderQuestionScreen();
    else if (state.screen === "review") renderReview();
    else renderResults();
  }
