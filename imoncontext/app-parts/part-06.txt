      window.setTimeout(() => document.querySelector(".answer-zone")?.classList.remove("shake"), 320);
      return;
    }

    if (index >= total - 1) {
      goToReview();
      return;
    }

    navigateToIndex(index + 1);
  }

  function navigateToIndex(index) {
    const questions = visibleQuestions();
    const clamped = Math.max(0, Math.min(questions.length - 1, index));
    state.currentQuestionId = questions[clamped]?.id || null;
    saveState();
    render();
    focusPrimaryControl();
  }

  function focusPrimaryControl() {
    requestAnimationFrame(() => {
      const target = document.querySelector("textarea, input.number-answer, .choice input, .scale-button");
      if (target && (target.tagName === "TEXTAREA" || target.classList.contains("number-answer"))) target.focus({ preventScroll: true });
    });
  }

  function goToReview() {
    const missing = visibleQuestions().filter((question) => question.required && !hasAnswer(question));
    if (missing.length) {
      state.currentQuestionId = missing[0].id;
      saveState();
      render();
      toast(`Осталось обязательных вопросов: ${missing.length}`);
      return;
    }
    state.screen = "review";
    saveState();
    render();
    window.scrollTo({ top: 0, behavior: "smooth" });
  }
